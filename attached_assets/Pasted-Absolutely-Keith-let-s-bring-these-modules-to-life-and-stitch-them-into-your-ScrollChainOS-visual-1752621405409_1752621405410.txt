```js
import { Web3Storage } from 'web3.storage';

const client = new Web3Storage({ token: 'YOUR_Web3Storage_API_Token' });

async function uploadToIPFS(fileBuffer, fileName) {
  const blob = new Blob([fileBuffer]);
  const file = new File([blob], fileName);
  const cid = await client.put([file]);
  return cid;
}
```

```js
async function verifyKey(apiKey, service) {
  let endpoint = '';
  if (service === 'openai') {
    endpoint = 'https://api.openai.com/v1/models';
  } else if (service === 'supabase') {
    endpoint = 'https://YOUR_PROJECT.supabase.co/rest/v1/';
  }

  try {
    const res = await fetch(endpoint, {
      headers: {
        Authorization: `Bearer ${apiKey}`
      }
    });
    return res.ok ? '✅ Valid Key' : '❌ Invalid';
  } catch {
    return '⚠️ Error checking key';
  }
}
```

## 🌌 Vault Scroll Signature Rendering

**📜 GitHub-based Scroll Trace**
```js
async function fetchCommitSigils(repo, filePath) {
  const res = await fetch(`https://api.github.com/repos/${repo}/commits?path=${filePath}`);
  const commits = await res.json();
  return commits.map(c => ({
    hash: c.sha,
    author: c.commit.author.name,
    date: c.commit.author.date
  }));
}
```

**🔮 Display Signature Glyphs**
