## ⚙️ `ScrollChain-Viewer.html` — Frontend Glyph Renderer (D3.js Enabled)

Save this in your project root as `ScrollChain-Viewer.html`:

```html
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8" />
  <title>ScrollChainOS Viewer</title>
  <script src="https://d3js.org/d3.v7.min.js"></script>
  <style>
    body { background: #0f0f0f; font-family: system-ui; color: #eee; padding: 20px; }
    h2 { color: #FFD700; border-bottom: 1px solid #444; }
    section { margin-bottom: 40px; }
    .glyph { background: #222; padding: 10px; margin: 5px 0; border-radius: 4px; }
    .circle { fill: #FFD700; stroke: #444; stroke-width: 2; }
    svg { margin-top: 20px; }
  </style>
</head>
<body>
  <h2>🌀 ScrollChainOS Visual Codex</h2>

  <section>
    <h3>📦 NFT Preview / Minting Codex</h3>
    <div id="nft-preview"></div>
  </section>

  <section>
    <h3>🧠 Kernel Entropy Curve</h3>
    <svg id="entropy-curve" width="600" height="200"></svg>
  </section>

  <section>
    <h3>🌍 DAO Scroll Signatures</h3>
    <div id="dao-sigils"></div>
  </section>

  <script>
    // Sample glyphs for NFT preview
    const nftData = [
      { name: "ProphecySigil", cid: "QmXyz..." },
      { name: "EntropyGlyph", cid: "QmAbc..." }
    ];
    nftData.forEach(item => {
      const div = document.createElement("div");
      div.className = "glyph";
      div.textContent = `🖼️ ${item.name} — CID: ${item.cid}`;
      document.getElementById("nft-preview").appendChild(div);
    });

    // Kernel entropy wave
    const kernelData = [25, 40, 60, 90];
    const line = d3.line()
      .x((d, i) => i * 120)
      .y(d => 150 - d)
      .curve(d3.curveBasis);
    const svg = d3.select("#entropy-curve");
    svg.append("path")
      .datum(kernelData.map((v, i) => ({ x: i, y: v })))
      .attr("fill", "none")
      .attr("stroke", "#FFD700")
      .attr("stroke-width", 3)
      .attr("d", line);

    // DAO signature glyphs
    const sigils = [
      { author: "Keith", hash: "0xfabc123", date: "2025-07-14T21:04:00Z" },
      { author: "PJ", hash: "0xfdef456", date: "2025-07-14T22:18:00Z" }
    ];
    sigils.forEach(sig => {
      const div = document.createElement("div");
      div.className = "glyph";
      div.textContent = `🧾 ${sig.author}: ${sig.hash.slice(0, 7)} on ${new Date(sig.date).toLocaleString()}`;
      document.getElementById("dao-sigils").appendChild(div);
    });
  </script>
</body>
</html>
```
