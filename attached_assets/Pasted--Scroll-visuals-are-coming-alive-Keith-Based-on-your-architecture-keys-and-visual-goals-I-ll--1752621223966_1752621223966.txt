## ⚙️ Starter Visual Renderer: `codex_render.html` + D3.js Setup

```html
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8" />
  <title>ScrollChainOS MetaCodex</title>
  <script src="https://d3js.org/d3.v7.min.js"></script>
  <style>
    body { font-family: system-ui; background: #111; color: #fdf6e3; margin: 0; padding: 20px; }
    .glyph { margin: 5px 0; padding: 8px; background: #222; border-radius: 5px; }
    svg { display: block; margin: 20px auto; }
    .title { font-size: 20px; margin-bottom: 10px; }
  </style>
</head>
<body>
  <div class="title">🌀 ScrollChainOS Glyph Stream</div>
  <div id="glyphs"></div>
  <svg width="400" height="300"></svg>

  <script>
    // Sample entropy glyphs from metadata
    const data = [
      { name: "ProphecyNode", value: 42 },
      { name: "EntropyThread", value: 73 },
      { name: "KernelAsset", value: 108 }
    ];

    // Render glyph cards
    d3.select("#glyphs")
      .selectAll(".glyph")
      .data(data)
      .enter()
      .append("div")
      .attr("class", "glyph")
      .text(d => "🔹 " + d.name + " — " + d.value);

    // Render circles as symbolic memory nodes
    const svg = d3.select("svg");
    svg.selectAll("circle")
      .data(data)
      .enter()
      .append("circle")
      .attr("cx", (d, i) => 100 + i * 100)
      .attr("cy", 150)
      .attr("r", d => Math.sqrt(d.value))
      .attr("fill", "#FFD700");
  </script>
</body>
</html>
```
